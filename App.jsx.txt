const UploadScreen = ({ go }) => {
  const [imgs, setImgs] = useState([
    { id: 1, url: "https://picsum.photos/200/200?random=10", ready: true },
    { id: 2, url: "https://picsum.photos/200/200?random=11", ready: true },
  ]);
  const [desc, setDesc] = useState("");
  const [loading, setLoading] = useState(false);

  const addImg = () => {
    const id = Date.now();
    setLoading(true);
    setImgs((p) => [...p, { id, url: `https://picsum.photos/200/200?random=${id}`, ready: false }]);
    setTimeout(() => {
      setImgs((p) => p.map((x) => (x.id === id ? { ...x, ready: true } : x)));
      setLoading(false);
    }, 1000);
  };

  return (
    <div className="h-full flex flex-col" style={{ background: T.bg }}>
      <div className="bg-white px-4 py-3 flex items-center gap-3 border-b border-gray-100">
        <button onClick={() => go("customer_home")} className="text-xl text-gray-500">←</button>

        <div className="grid grid-cols-2 gap-3">
          {imgs.map((img) => (
            <div key={img.id} className="bg-white rounded-2xl p-2 border border-gray-200 relative">
              <img src={img.url} alt="" className="w-full h-32 object-cover rounded-xl" />

      case "customer_home": return <CustomerHome go={setScreen} />;

      case "chat": return <ChatScreen go={setScreen} />;
      case "partner_portal": return <PartnerScreen go={setScreen} />;
      case "service_admin": return <ServiceAdminScreen go={setScreen} />;

      case "logistics_login": return <LogisticsLoginScreen go={setScreen} />;
      case "dashboard": return <ShipperDashboard go={setScreen} />;
      case "packing3d": return <Packing3DScreen go={setScreen} />;
      case "driver": return <DriverScreen go={setScreen} />;
      case "logistics_admin": return <AdminLogisticsDash go={setScreen} />;

      default: return <EntryScreen go={setScreen} />;
    }
  };

  return (
    <div className="min-h-screen bg-gray-950 flex items-center justify-center p-0 sm:p-4">
      <div className="w-full sm:max-w-md h-screen sm:h-[840px] bg-white sm:rounded-[40px] sm:shadow-2xl overflow-hidden relative flex flex-col border border-gray-200">
