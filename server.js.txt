import express from 'express';
import path from 'path';
import { fileURLToPath } from 'url';
import { PrismaClient } from '@prisma/client';

const app = express();
const prisma = new PrismaClient();
const PORT = process.env.PORT || 3000;

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

app.use(express.json());

app.get('/api/health', async (req, res) => {
  try {
    await prisma.$queryRaw`SELECT 1`;
    res.json({
      ok: true,
      message: 'Server and database are running'
    });
  } catch (error) {
    console.error('Health check error:', error);
    res.status(500).json({
      ok: false,
      message: 'Database connection failed'
    });
  }
});

app.get('/api/stats', async (req, res) => {
  try {
    const [total, pending, inTransit, delivered] = await Promise.all([
      prisma.shipment.count(),
      prisma.shipment.count({ where: { status: 'Pending' } }),
      prisma.shipment.count({ where: { status: 'In Transit' } }),
      prisma.shipment.count({ where: { status: 'Delivered' } })
    ]);

    res.json({
      total,
      pending,
      inTransit,
      delivered
    });
  } catch (error) {
    console.error('Stats error:', error);
    res.status(500).json({ error: 'ไม่สามารถโหลดสถิติได้' });
  }
});

app.get('/api/shipments', async (req, res) => {
  try {
    const shipments = await prisma.shipment.findMany({
      orderBy: {
        createdAt: 'desc'
      }
    });
    res.json(shipments);
  } catch (error) {
    console.error('Fetch shipments error:', error);
    res.status(500).json({ error: 'ไม่สามารถโหลดข้อมูล shipment ได้' });
  }
});

app.post('/api/shipments', async (req, res) => {
  try {
    const { customer, origin, destination, status, vehicle, eta } = req.body;

    if (!customer || !origin || !destination || !status || !vehicle || !eta) {
      return res.status(400).json({
        error: 'กรุณากรอกข้อมูลให้ครบทุกช่อง'
      });
    }

    const count = await prisma.shipment.count();
    const trackingNo = `TH${String(100001 + count).padStart(6, '0')}`;

    const shipment = await prisma.shipment.create({
      data: {
        trackingNo,
        customer,
        origin,
        destination,
        status,
        vehicle,
        eta: new Date(eta)
      }
    });

    res.status(201).json(shipment);
  } catch (error) {
    console.error('Create shipment error:', error);
    res.status(500).json({ error: 'ไม่สามารถสร้าง shipment ได้' });
  }
});

app.put('/api/shipments/:id/status', async (req, res) => {
  try {
    const id = Number(req.params.id);
    const { status } = req.body;

    if (!id || !status) {
      return res.status(400).json({ error: 'ข้อมูลไม่ถูกต้อง' });
    }

    const existing = await prisma.shipment.findUnique({
      where: { id }
    });

    if (!existing) {
      return res.status(404).json({ error: 'ไม่พบ shipment' });
    }

    const updated = await prisma.shipment.update({
      where: { id },
      data: { status }
    });

    res.json(updated);
  } catch (error) {
    console.error('Update status error:', error);
    res.status(500).json({ error: 'ไม่สามารถอัปเดตสถานะได้' });
  }
});

const distPath = path.join(__dirname, 'dist');
app.use(express.static(distPath));

app.get('*', (req, res) => {
  res.sendFile(path.join(distPath, 'index.html'));
});

async function startServer() {
  try {
    await prisma.$connect();
    app.listen(PORT, () => {
      console.log(`Server running on port ${PORT}`);
    });
  } catch (error) {
    console.error('Failed to start server:', error);
    process.exit(1);
  }
}

startServer();

process.on('SIGINT', async () => {
  await prisma.$disconnect();
  process.exit(0);
});

process.on('SIGTERM', async () => {
  await prisma.$disconnect();
  process.exit(0);
});
