generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model Shipment {
  id          Int      @id @default(autoincrement())
  trackingNo  String   @unique
  customer    String
  origin      String
  destination String
  status      String
  vehicle     String
  eta         DateTime
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
}
